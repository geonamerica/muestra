01 · Kit de herramientas

Las librerías que vas a usar

No hace falta reinventar nada. Estas cuatro piezas cubren todo lo que necesita un editor de PDF.

⚡

PyMuPDF

El motor principal. Lee, renderiza y modifica PDFs a toda velocidad: texto, anotaciones, imágenes y páginas.

🖼️

Pillow

Convierte cada página renderizada en una imagen que la interfaz gráfica puede mostrar en pantalla.

🪟

Tkinter

Ya viene con Python. Botones, lienzo y diálogos para abrir y guardar: la cara visible de tu editor.

🧩

pypdf (opcional)

Alternativa 100% Python, ideal para tareas simples como unir, dividir o encriptar sin dependencias nativas.

Abrir y leer un PDF

Un documento se comporta como una lista de páginas. Podés recorrerlo, consultar sus metadatos y extraer el texto de cada página.

leer.py
import pymupdf

doc = pymupdf.open("contrato.pdf")
print(f"Páginas: {doc.page_count}")
print(doc.metadata)

for page in doc:
    texto = page.get_text()
    print(f"--- Página {page.number + 1} ---")
    print(texto[:200])

Escribir, resaltar y censurar

Insertá texto en cualquier coordenada, buscá palabras para resaltarlas y tapá datos sensibles con redacciones reales (el texto se elimina del archivo, no solo se oculta).

editar.py
import pymupdf

doc = pymupdf.open("contrato.pdf")
page = doc[0]

# Escribir texto nuevo (x=72, y=72 puntos desde arriba a la izquierda)
page.insert_text((72, 72), "APROBADO", fontsize=28, color=(0, 0.6, 0.3))

# Buscar una palabra y resaltarla
for rect in page.search_for("importe"):
    page.add_highlight_annot(rect)

# Censurar datos sensibles de verdad
for rect in page.search_for("DNI"):
    page.add_redact_annot(rect, fill=(0, 0, 0))
page.apply_redactions()

doc.save("contrato_editado.pdf")

Marca de agua e imágenes

Recorré todas las páginas y agregá una marca de agua semitransparente inclinada. También podés estampar tu logo en una esquina.

marca_agua.py
import pymupdf

doc = pymupdf.open("contrato.pdf")

for page in doc:
    punto = pymupdf.Point(page.rect.width * 0.18, page.rect.height * 0.6)
    page.insert_text(
        punto, "CONFIDENCIAL",
        fontsize=64, color=(1, 0, 0.4),
        fill_opacity=0.18,
        morph=(punto, pymupdf.Matrix(35)),  # inclinar 35 grados
    )
    # Logo en la esquina superior derecha
    page.insert_image(pymupdf.Rect(480, 20, 570, 80), filename="logo.png")

doc.save("contrato_marca.pdf")

Unir, dividir y rotar páginas

Combiná varios archivos en uno, extraé un rango de páginas o reorganizá el documento como quieras.

paginas.py
import pymupdf

# Unir varios PDFs
final = pymupdf.open()
for ruta in ["a.pdf", "b.pdf", "c.pdf"]:
    with pymupdf.open(ruta) as parte:
        final.insert_pdf(parte)
final.save("unido.pdf")

# Dividir: quedarse con las páginas 1 a 3
doc = pymupdf.open("unido.pdf")
extracto = pymupdf.open()
extracto.insert_pdf(doc, from_page=0, to_page=2)
extracto.save("primeras_3.pdf")

# Rotar la primera página y borrar la última
doc[0].set_rotation(90)
doc.delete_page(-1)
doc.save("reorganizado.pdf")

Mostrarlo en pantalla

La clave de la interfaz: renderizar la página como imagen con get_pixmap y dibujarla en un lienzo de Tkinter. El editor completo está más abajo.

visor.py
import tkinter as tk
import pymupdf
from PIL import Image, ImageTk

doc = pymupdf.open("contrato.pdf")
pix = doc[0].get_pixmap(matrix=pymupdf.Matrix(1.5, 1.5))  # zoom 150%
img = Image.frombytes("RGB", (pix.width, pix.height), pix.samples)

root = tk.Tk()
foto = ImageTk.PhotoImage(img)
tk.Label(root, image=foto).pack()
root.mainloop()

04 · Código completo

Tu editor PDF, listo para usar

Abre archivos, navega páginas, escribe texto con un clic, resalta o censura arrastrando el mouse, rota y guarda. Menos de 100 líneas.

editor_pdf.py
import tkinter as tk
from tkinter import filedialog, simpledialog
import pymupdf
from PIL import Image, ImageTk


class EditorPDF:
    def __init__(self, root):
        self.root = root
        self.root.title("Mi Editor PDF")
        self.doc = None
        self.pagina = 0
        self.zoom = 1.5
        self.modo = None  # "texto", "resaltar" o "censurar"
        self.inicio = None

        barra = tk.Frame(root, bg="#1e1e2e")
        barra.pack(fill="x")
        botones = [
            ("Abrir", self.abrir),
            ("◀", self.anterior),
            ("▶", self.siguiente),
            ("Texto", lambda: self.set_modo("texto")),
            ("Resaltar", lambda: self.set_modo("resaltar")),
            ("Censurar", lambda: self.set_modo("censurar")),
            ("Rotar", self.rotar),
            ("Guardar", self.guardar),
        ]
        for texto, comando in botones:
            tk.Button(barra, text=texto, command=comando).pack(side="left", padx=4, pady=4)

        self.canvas = tk.Canvas(root, bg="#2a2a3a", highlightthickness=0)
        self.canvas.pack(fill="both", expand=True)
        self.canvas.bind("<ButtonPress-1>", self.click)
        self.canvas.bind("<ButtonRelease-1>", self.soltar)

    # ---------- archivo ----------
    def abrir(self):
        ruta = filedialog.askopenfilename(filetypes=[("PDF", "*.pdf")])
        if ruta:
            self.doc = pymupdf.open(ruta)
            self.pagina = 0
            self.render()

    def guardar(self):
        if not self.doc:
            return
        ruta = filedialog.asksaveasfilename(defaultextension=".pdf")
        if ruta:
            self.doc.save(ruta, garbage=3, deflate=True)

    # ---------- vista ----------
    def render(self):
        page = self.doc[self.pagina]
        pix = page.get_pixmap(matrix=pymupdf.Matrix(self.zoom, self.zoom))
        img = Image.frombytes("RGB", (pix.width, pix.height), pix.samples)
        self.foto = ImageTk.PhotoImage(img)
        self.canvas.delete("all")
        self.canvas.create_image(0, 0, anchor="nw", image=self.foto)
        self.root.title(f"Mi Editor PDF · página {self.pagina + 1}/{len(self.doc)}")

    def anterior(self):
        if self.doc and self.pagina > 0:
            self.pagina -= 1
            self.render()

    def siguiente(self):
        if self.doc and self.pagina < len(self.doc) - 1:
            self.pagina += 1
            self.render()

    # ---------- edición ----------
    def set_modo(self, modo):
        self.modo = modo

    def a_pdf(self, x, y):
        # Convierte píxeles del lienzo en coordenadas de la página
        page = self.doc[self.pagina]
        return pymupdf.Point(x / self.zoom, y / self.zoom) * page.derotation_matrix

    def click(self, evento):
        if not self.doc or not self.modo:
            return
        self.inicio = self.a_pdf(evento.x, evento.y)
        if self.modo == "texto":
            texto = simpledialog.askstring("Texto", "¿Qué querés escribir?")
            if texto:
                self.doc[self.pagina].insert_text(self.inicio, texto, fontsize=14, color=(0.1, 0.2, 0.8))
                self.render()

    def soltar(self, evento):
        if not self.doc or self.modo not in ("resaltar", "censurar") or not self.inicio:
            return
        page = self.doc[self.pagina]
        rect = pymupdf.Rect(self.inicio, self.a_pdf(evento.x, evento.y)).normalize()
        if rect.is_empty:
            return
        if self.modo == "resaltar":
            page.add_highlight_annot(rect)
        else:
            page.add_redact_annot(rect, fill=(0, 0, 0))
            page.apply_redactions()
        self.render()

    def rotar(self):
        if self.doc:
            page = self.doc[self.pagina]
            page.set_rotation((page.rotation + 90) % 360)
            self.render()


if __name__ == "__main__":
    root = tk.Tk()
    root.geometry("950x1000")
    EditorPDF(root)
    root.mainloop()
